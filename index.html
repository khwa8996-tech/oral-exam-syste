import os import pandas as pd from docx import Document from docx.shared import Inches, Pt, RGBColor from docx.enum.text import WD_ALIGN_PARAGRAPH def generate_smart_ai_system(): doc = Document() # ضبط هوامش الصفحة for section in doc.sections: section.top_margin = Inches(1) section.bottom_margin = Inches(1) section.left_margin = Inches(1) section.right_margin = Inches(1) # عنوان النظام الذكي title_p = doc.add_paragraph() title_p.paragraph_format.rtl = True title_p.paragraph_format.alignment = WD_ALIGN_PARAGRAPH.CENTER title_run = title_p.add_run("بنك أسئلة امتحان مادة الحاسوب (النظام الذكي المتطور)") title_run.font.name = 'Arial' title_run.font.size = Pt(20) title_run.font.bold = True title_run.font.color.rgb = RGBColor(15, 23, 42) subtitle_p = doc.add_paragraph() subtitle_p.paragraph_format.rtl = True subtitle_p.paragraph_format.alignment = WD_ALIGN_PARAGRAPH.CENTER subtitle_run = subtitle_p.add_run("أسئلة اختيار من متعدد مع التظليل البرمجي الذكي للإجابات الصحيحة") subtitle_run.font.name = 'Arial' subtitle_run.font.size = Pt(13) subtitle_run.font.color.rgb = RGBColor(71, 85, 105) doc.add_paragraph().paragraph_format.space_after = Pt(12) # قاعدة البيانات الشاملة للأسئلة questions_data = [ ("س 1: ما هو مفهوم تطبيقات الأجهزة الذكية؟", [ ("أ", "برامج مخصصة لتشغيل الأجهزة المكتبية فقط"), ("ب", "برامج مكتوبة بلغات برمجية تتوافق مع أنظمة تشغيل الأجهزة الذكية وتقدم خدمات في شتى المجالات (الإجابة الصحيحة)"), ("ج", "ملفات نصية لتخزين البيانات فقط"), ("د", "أجهزة مادية لإدارة الاتصال بالإنترنت") ]), # ... بقية الأسئلة الـ 107 ... ] for idx, (q_text, opts) in enumerate(questions_data, 1): p_q = doc.add_paragraph() p_q.paragraph_format.rtl = True p_q.paragraph_format.alignment = WD_ALIGN_PARAGRAPH.RIGHT p_q.paragraph_format.space_before = Pt(16) p_q.paragraph_format.space_after = Pt(6) p_q.paragraph_format.keep_with_next = True run_q = p_q.add_run(q_text) run_q.font.name = 'Arial' run_q.font.size = Pt(12) run_q.font.bold = True run_q.font.color.rgb = RGBColor(15, 23, 42) for opt_letter, opt_text in opts: p_opt = doc.add_paragraph() p_opt.paragraph_format.rtl = True p_opt.paragraph_format.alignment = WD_ALIGN_PARAGRAPH.RIGHT p_opt.paragraph_format.space_before = Pt(0) p_opt.paragraph_format.space_after = Pt(4) p_opt.paragraph_format.right_indent = Inches(0.3) run_opt = p_opt.add_run(f"{opt_letter}) {opt_text}") run_opt.font.name = 'Arial' run_opt.font.size = Pt(11) # خوارزمية الذكاء الاصطناعي للتظليل التلقائي للإجابة الصحيحة if "(الإجابة الصحيحة)" in opt_text: run_opt.font.bold = True run_opt.font.color.rgb = RGBColor(220, 38, 38) # تظليل بلون مميز else: run_opt.font.color.rgb = RGBColor(51, 65, 85) doc.save("Smart_AI_Questions_System-v2.docx") print("تم معالجة الأسئلة وتوليد الملف الذكي بنجاح!") generate_smart_ai_system()
